### 3. TreatmentLog Model (NEW)
```typescript
// server/src/modules/treatment/treatmentLogModel.ts
import mongoose, { Schema, Document } from 'mongoose';

export interface ITreatmentLog extends Document {
  logId: string;
  treatmentPlanId: string;
  patientId: string;
  date: Date;                          // Date of log entry (local date)
  medicineLogs: IMedicineLogEntry[];
  symptomLog?: ISymptomLogEntry;

  photoProgress?: IPhotoProgressEntry;
  adherencePercentage: number;         // 0-100 for the day
  notes?: string;
  simulation: true;
  createdAt: Date;
  updatedAt: Date;
}

export interface IMedicineLogEntry {
  medicineId: string;
  medicineName: string;
  scheduledTime: string;               // "08:00"
  takenTime?: string;                  // Actual time taken
  taken: boolean;
  skipped: boolean;
  skipReason?: string;
}

export interface ISymptomLogEntry {
  symptoms: string[];                  // ["burning", "swelling", "difficulty swallowing"]
  severity: 'none' | 'mild' | 'moderate' | 'severe';
  notes?: string;
}

export interface IPainLogEntry {
  score: number;                       // 0-10 VAS scale
  location?: string;
  notes?: string;
}

export interface IPhotoProgressEntry {
  photoId: string;                     // References ScreeningPhoto.photoId
  fileId?: string;                     // Authenticated private storage file ID
  description?: string;
  anatomicalSite?: string;
}

const TreatmentLogSchema = new Schema<ITreatmentLog>({
  logId: { type: String, required: true, unique: true, index: true },
  treatmentPlanId: { type: String, required: true, index: true },
  patientId: { type: String, required: true, index: true },
  date: { type: Date, required: true, index: true },
  medicineLogs: [{
    medicineId: { type: String, required: true },
    medicineName: { type: String, required: true },
    scheduledTime: { type: String, required: true },
    takenTime: { type: String },
    taken: { type: Boolean, default: false },
    skipped: { type: Boolean, default: false },
    skipReason: { type: String },
  }],
  symptomLog: {
    symptoms: [{ type: String }],
    severity: { type: String, enum: ['none', 'mild', 'moderate', 'severe'] },
    notes: { type: String },
  },
  painLog: {
    score: { type: Number, min: 0, max: 10 },
    location: { type: String },
    notes: { type: String },
  },
  photoProgress: {
    photoId: { type: String },
    fileId: { type: String },
    description: { type: String },
    anatomicalSite: { type: String },
  },
  adherencePercentage: { type: Number, default: 0, min: 0, max: 100 },
  notes: { type: String },
  simulation: { type: Boolean, default: true, immutable: true },

// Compound index for daily log lookup
TreatmentLogSchema.index({ treatmentPlanId: 1, date: 1 }, { unique: true });

export const TreatmentLog = mongoose.models.TreatmentLog || mongoose.model<ITreatmentLog>('TreatmentLog', TreatmentLogSchema);
```

## New API Endpoints (REST, `/api/patient/*` prefix)

### Module 1: Registration & Login
| Method | Path | Auth | Description |
|--------|------|------|-------------|
| POST | `/api/patient/auth/register` | Public | Register a patient with explicit consent; registration does not require OTP and creates User + Patient records |
| POST | `/api/patient/auth/login` | Public | Patient login (email/phone + password OR OTP); returns JWT |
| POST | `/api/patient/auth/forgot-password` | Public | Initiate OTP-based password reset |
| POST | `/api/patient/auth/reset-password` | Public | Complete password reset with OTP |
| GET | `/api/patient/auth/me` | Patient | Get current patient profile (reuse `/api/v1/auth/me`) |

### Module 2: Dashboard
| Method | Path | Auth | Description |
|--------|------|------|-------------|
| GET | `/api/patient/dashboard/summary` | Patient | Aggregated summary: next appointment, latest screening, active treatment, unread reminders, recent reports |

### Module 3: Personal Health Profile
| Method | Path | Auth | Description |
|--------|------|------|-------------|
| GET | `/api/patient/profile` | Patient | Get extended profile (demographics, medical history, allergies, medications, habits, emergency contact, insurance, language, completeness %) |
| PUT | `/api/patient/profile` | Patient | Update profile fields |
| GET | `/api/patient/profile/completeness` | Patient | Get profile completeness breakdown |

### Module 4: Oral Screening & AI Report
| Method | Path | Auth | Description |
|--------|------|------|-------------|
| POST | `/api/patient/screening/analyze` | Patient | Upload image/symptoms → AI analysis (reuse `/api/ai/analyze` + `/api/v1/screening/ai/analyze`) |
| GET | `/api/patient/screening/history` | Patient | List patient's screenings (reuse `/api/ai/screenings?patientId=`) |
| GET | `/api/patient/screening/:screeningId/report` | Patient | Get AI report detail (findings, risk, plain-language, disclaimer) |
| POST | `/api/patient/screening/:screeningId/chat` | Patient | Send message in AI chat for this screening |
| GET | `/api/patient/screening/:screeningId/chat` | Patient | Get chat history for this screening |

### Module 5: AI Assessment & Referrals
| Method | Path | Auth | Description |
|--------|------|------|-------------|
| POST | `/api/patient/assessment/:screeningId` | Patient | Generate assessment from screening (condition→specialty, doctor suggestions, hospital referrals) |
| GET | `/api/patient/assessment/:screeningId` | Patient | Get assessment detail |

## Authorization & Scoping Rules
- All `/api/patient/*` routes require `requireAuth` + `role === 'patient'`
- Patient can only access own data: `patientId === req.user.patientId`
- Cross-hospital denial enforced (Invariant 7)
- Consent required before sharing screening/profile with doctor (Module 6)
- Audit logging on all mutations (`auditTag` middleware)
- Idempotency via `clientMutationId` on all create endpoints
